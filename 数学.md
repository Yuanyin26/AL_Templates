```c++
            for(int k = 1;k <= len;k++)
```

## 中国剩余定理（CRT,exCRT）

```c++
namespace CRT
{
    int exgcd(int a, int b, int &x, int &y)
    {
        if (!b) 
        {
            x = 1;
            y = 0;
            return a;
        }
        int d = exgcd(b, a % b, x, y);
        int t = x;
        x = y;
        y = t - (a / b) * y;
        return d;
    }
    //返回通解 x === ai mod mi [0,n) m 两两互质
    pii CRT(int n,std::vector<int> &a,std::vector<int> &m)
    {
        int M = 1;
        for(int i = 0;i < n;i++)M = M * m[i];
        i128 ans = 0;
        for(int i = 0;i < n;i++)
        {
            int w = M / m[i];
            int inv,y;inv = y = 0; // The inv of w
            exgcd(w,m[i],inv,y);
            inv = (inv % m[i] + m[i]) % m[i];
            ans += i128(a[i]) * w % M * inv % M;
            ans %= M;
        }
        return {ans,M};
    }
    //返回通解 x === ai mod mi [0,n) m 两两不一定互质,{-1,-1} 为无解
    pii exCRT(int n,std::vector<int> &a,std::vector<int> &m)
    {
        int a1 = a[0],m1 = m[0];
        for(int i = 1;i < n;i++)
        {
            int x,y;
            int g = exgcd(m1,m[i],x,y);//gcd(m1,m[i])
            if((a[i] - a1) % g != 0)return {-1,-1};
            i128 P = (i128)x * ((a[i] - a1) / g);//特解，没有限制正负

            int M = m1 / g * m[i];//新的 m1

            a1 = (P * m1 % M + a1) % M;//新的 a1 = m1 * x + a1
            a1 = (a1 % M + M) % M;
            m1 = M;
        }
        return {a1,m1};
    }
};
```

```c++
struct Lucas
```

## exLucas定理

```c++
struct exLucas
{
    struct node
    {
        int p,k,q;
        std::vector<int>c,s,f,f_inv;
        bool fl;
        int qmi(int a,int b)
        {
            int res = 1;
            while(b)
            {
                if(b & 1)res = res * a % q;
                a = a * a % q;
                b >>= 1;
            }
            return res;
        }
        node(int p,int k,int q,int n) : p(p),k(k),q(q),s(q),fl(0)
        {
            s[0] = 1;
            for(int i = 1;i < q;i++)
                s[i] = s[i - 1] * (i % p ? i : 1) % q;//分块，前 i 项积
            if(n <= N)
            {
                fl = 1;
                c.resize(n + 1);
                f.resize(n + 1);
                f_inv.resize(n + 1);
                f[0] = 1;
                for(int i = 1;i <= n;i++)
                {
                    c[i] = c[i - 1];
                    int cnt = i;
                    while(cnt % p == 0)c[i]++,cnt /= p;
                    f[i] = f[i - 1] * cnt % q;
                }

                f_inv[n] = qmi(f[n],q - q/p - 1);//phi(q) - 1
                for(int i = n;i >= 1;i--)
                {
                    int x = i;
                    while(x % p == 0)x /= p;
                    f_inv[i - 1] = f_inv[i] * x % q; 
                }
            }
        }
        int F_c(int n)
        {
            int ans = 0;
            while(n)
            {
                ans += n / p;
                n /= p;
            }
            return ans;
        }
        int F_f(int n)
        {
            if(n == 0)return 1;
            int ans = qmi(s[q - 1],n / q) * s[n % q] % q * F_f(n / p) % q;
            return ans;
        }
        int F_f_inv(int n)
        {
            int ans = F_f(n);
            return qmi(ans,q - q / p - 1);
        }

        //n <= N,使用预处理，否则现算
        pii exlucas(int n,int m)
        {
            if(m > n)return {0,q};
            if(fl)
            {
                int cnt = c[n] - c[m] - c[n - m];
                int ans = qmi(p,cnt);
                int tmp = f[n] * f_inv[n - m] % q * f_inv[m] % q;
                return {ans * tmp % q,q};
            }
            else
            {
                int cnt = F_c(n) - F_c(m) - F_c(n - m);
                int ans = qmi(p,cnt);
                int tmp = F_f(n) * F_f_inv(n - m) % q * F_f_inv(m) % q;
                return {ans * tmp % q,q};
            }
        }
    };
    int M,siz;//M <= N;
    std::vector<node>pri;//每个质因数预处理
    //1 ~ n 范围组合数，对 M 取余
    exLucas(int m,int n) : M(m),siz(0)
    {
        while(m != 1)
        {
            int p = minp[m],k = 0,q = 1;
            while(minp[m] == p)
            {
                k++;
                q *= p;
                m /= p;
            }
            pri.push_back(node{p,k,q,n});
            siz++;
        }
    }
    int C(int n,int m)
    {
        if(m > n)return 0;
        std::vector<int>a(siz),b(siz);
        for(int i = 0;i < siz;i++)
        {
            pii ans = pri[i].exlucas(n,m);
            a[i] = ans.first;
            b[i] = ans.second;
        }
        return CRT::CRT(siz,a,b).first;
    }
};
```

```c++
bool Miller_Rabin(int p)//判是否是素数
{
    if(p < 2)return false;
    if(p == 2 || p == 3)return true;
    int d = p - 1,r = 0;
    while(!(d & 1))r++,d >>= 1;  // 将d处理为奇数
    for(int k = 0;k < 10;k++)
    {
        int a = rand() % (p - 2) + 2;
        int x = qmi(a, d, p);
        if (x == 1 || x == p - 1) continue;
        for(int i = 0;i < r - 1;i++)
        {
            x = mul(x,x,p);
            if(x == p - 1)break;
        }
        if (x != p - 1) return false;
    }
    return true;

int Pollard_Rho(int x)//取x的一个因数
{
    int s = 0, t = 0;
    int c = (int)rand() % (x - 1) + 1;
    int step = 0,goal = 1;
    int val = 1;
    for(goal = 1;;goal <<= 1,s = t,val = 1)
    {
        for(step = 1;step <= goal;step++)
        {

            t = (mul(t,t,x) + c) % x;
            val = mul(val,llabs(t - s),x);
            if(step % 127 == 0)
            {
                int d = gcd(val,x);
                if(d > 1)return d;
            }
        }
        int d = gcd(val,x);
        if(d > 1)return d;
    }
}

{
    if(x <= mx || x < 2)return;
    if(Miller_Rabin(x))
    {
        mx = max(mx,x);
        return;
    }
    int p = x;
    while(p >= x)p = Pollard_Rho(x);
    while(x % p == 0)x /= p;

}
```

## 快速傅里叶变换(FFT)

```c++
typedef complex<double> cp;
const double pi = acos(-1);
vector<int>rev;//位逆序置换
void fft(vector<cp> &a,int flag = 1)//1为DFT,-1为IDFT
{
    int n = a.size();
    if(rev.size() != n)
    {
        rev.resize(n);
        int k = __builtin_ctz(n) - 1;
        for(int i = 0;i < n;i++)
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << k;
    }
    for(int i = 0;i < n;i++)
        if(rev[i] < i)swap(a[i],a[rev[i]]);
    for(int len = 1;len < n;len *= 2)
    {
        cp wn = exp(cp(0,flag * pi / len));//单位根
        for(int i = 0;i < n;i += 2 * len)//合并每一组
        {
            cp w = cp(1,0);//w0
            for(int j = 0;j < len;j++)//合并
            {
                cp x = a[i + j];
                cp y = a[i + j + len] * w;
                a[i + j] = x + y;
                a[i + j + len] = x - y;
                w *= wn;//wi + 1;
            }
        }
    }
    if(flag == -1)
        for(int i = 0;i < n;i++)
            a[i] /= n;
}
vector<cp> mul(vector<cp> a,vector<cp> b)//求卷积
{
    int n = 1,tot = a.size() + b.size() - 1;
    while(n < tot)n <<= 1;

    b.resize(n);
    fft(a);
    fft(b);
    for(int i = 0;i < n;i++)a[i] = a[i] * b[i];
    fft(a,-1);
    a.resize(tot);//x ^ (tot - 1)
    return a;
}
```

## 快速数论变换(NTT)

```c++

int qmi(int a,int b)
{

    while(b)
    {
        if(b & 1)res = res * a % mod;
        b >>= 1;
        a = a * a % mod;
    }
    return res;
}
vector<int>rev;
void ntt(vector<int> &a,int flag = 1)
{
    int n = a.size();
    if(rev.size() != n)
    {
        rev.resize(n);
        int k = __builtin_ctz(n) - 1;
        for(int i = 0;i < n;i++)
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << k;
    }
    for(int i = 0;i < n;i++)
        if(rev[i] < i)swap(a[i],a[rev[i]]);
    for(int len = 1;len < n;len *= 2)
    {
        int gn = qmi(3,(mod - 1) / (len << 1));//原根
        for(int i = 0;i < n;i += 2 * len)
        {
            int g = 1;
            for(int j = 0;j < len;j++)
            {
                int x = a[i + j];
                int y = a[i + j + len] * g % mod;
                a[i + j] = (x + y) % mod;
                a[i + j + len] = (x - y + mod) % mod;
                g = g * gn % mod;
            }
        }
    }
    if(flag == -1)
    {
        reverse(a.begin() + 1,a.end());//多个交换
        int inv = qmi(n,mod - 2);

    }
}
```

```c++
int qmi(int a,int b)
{
    int res = 1;
    while(b)
    {
        if(b & 1)res = res * a % mod;
        b >>= 1;
        a = a * a % mod;
    }
    return res;
}
int add(int u,int v)
{
    return (u + v + mod) % mod;
}

void fwt_or(vector<int>&a,int flag = 1)//-1为逆变换
{
    int n = a.size();

        for(int i = 0;i < n;i += len * 2)
            for(int j = 0;j < len;j++)
                a[i + j + len] = add(a[i + j + len],mul(flag,a[i + j]));
    return;
}
```

```c++
template<class T>
constexpr T power(T a, u64 b, T res = 1) {
    for (; b != 0; b /= 2, a *= a) {
        if (b & 1) {
            res *= a;
        }
    }
    return res;
}

template<u32 P>
struct ModIntBase {
public:
    constexpr ModIntBase() : x(0) {}
    template<class T>
    constexpr ModIntBase(T x_) {
        T v = x_ % T(mod());
        if (v < 0) {
            v += mod();
        }
        x = v;
    }
    
    constexpr static u32 mod() {
        return P;
    }
    
    constexpr u32 val() const {
        return x;
    }

    constexpr u32 mulMod(u32 a, u32 b) {
        return u64(a) * b % mod();
    }
    
    constexpr ModIntBase operator-() const {
        ModIntBase res;
        res.x = (x == 0 ? 0 : mod() - x);
        return res;
    }
    
    constexpr ModIntBase inv() const {
        return power(*this, mod() - 2);
    }
    
    constexpr ModIntBase &operator*=(const ModIntBase &rhs) & {
        x = mulMod(x, rhs.val());
        return *this;
    }
    constexpr ModIntBase &operator+=(const ModIntBase &rhs) & {
        x += rhs.val();
        if (x >= mod()) {
            x -= mod();
        }
        return *this;
    }
    constexpr ModIntBase &operator-=(const ModIntBase &rhs) & {
        x -= rhs.val();
        if (x >= mod()) {
            x += mod();
        }
        return *this;
    }
    constexpr ModIntBase &operator/=(const ModIntBase &rhs) & {
        return *this *= rhs.inv();
    }
    
    friend constexpr ModIntBase operator*(ModIntBase lhs, const ModIntBase &rhs) {
        lhs *= rhs;

constexpr int P = 998244353;
using Z = ModInt<P>;
std::vector<int> rev;
std::vector<Z> roots{0, 1};
void dft(std::vector<Z> &a) {
    int n = a.size();
    
    if (int(rev.size()) != n) {
        int k = __builtin_ctz(n) - 1;
        rev.resize(n);
        for (int i = 0; i < n; i++) {
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << k;
        }
    }
    
    for (int i = 0; i < n; i++) {
        if (rev[i] < i) {
            std::swap(a[i], a[rev[i]]);
        }
    }
    if (int(roots.size()) < n) {
        int k = __builtin_ctz(roots.size());
        roots.resize(n);
        while ((1 << k) < n) {
            Z e = power(Z(3), (P - 1) >> (k + 1));
            for (int i = 1 << (k - 1); i < (1 << k); i++) {
                roots[2 * i] = roots[i];
                roots[2 * i + 1] = roots[i] * e;
            }
            k++;
        }
    }
    for (int k = 1; k < n; k *= 2) {
        for (int i = 0; i < n; i += 2 * k) {
            for (int j = 0; j < k; j++) {
                Z u = a[i + j];
                Z v = a[i + j + k] * roots[k + j];
                a[i + j] = u + v;
                a[i + j + k] = u - v;
            }
        }
    }
}
void idft(std::vector<Z> &a) {
    int n = a.size();
    std::reverse(a.begin() + 1, a.end());
    dft(a);
    Z inv = (1 - P) / n;
    for (int i = 0; i < n; i++) {
        a[i] *= inv;
    }
}
struct Poly {
    std::vector<Z> a;
    Poly() {}
    explicit Poly(int size, std::function<Z(int)> f = [](int) { return 0; }) : a(size) {
        for (int i = 0; i < size; i++) {
            a[i] = f(i);
        }
    }

            for (int i = 0; i < a.size(); i++) {
                for (int j = 0; j < b.size(); j++) {
                    c[i + j] += a[i] * b[j];

            return c;

            sz *= 2;

        a.a.resize(sz);
        b.a.resize(sz);

        dft(b.a);
        for (int i = 0; i < sz; ++i) {
            a.a[i] = a[i] * b[i];

            return Poly();
        }

            res[i] = (i + 1) * a[i + 1];

        return Poly(res);

            res[i + 1] = a[i] / (i + 1);

        int k = 1;
        while (k < m) {
            k *= 2;
            x = (x * (Poly{2} - modxk(k) * x)).modxk(k);
        }
        return x.modxk(m);
    }

        return (deriv() * inv(m)).integr().modxk(m);
    }
    Poly exp(int m) const {
        Poly x{1};
        int k = 1;
        while (k < m) {
            k *= 2;
            x = (x * (Poly{1} - x.log(k) + modxk(k))).modxk(k);
        }
        return x.modxk(m);
    }
    Poly pow(int k, int m) const {
        int i = 0;
        while (i < size() && a[i].val() == 0) {
            i++;
        }
        if (i == size() || 1LL * i * k >= m) {
            return Poly(std::vector<Z>(m));
        }
        Z v = a[i];
        auto f = divxk(i) * v.inv();
        return (f.log(m - i * k) * k).exp(m - i * k).mulxk(i * k) * power(v, k);
    }
    Poly sqrt(int m) const {
        Poly x{1};
        int k = 1;
        while (k < m) {
            k *= 2;
            x = (x + (modxk(k) * x.inv(k)).modxk(k)) * ((P + 1) / 2);
        }
        return x.modxk(m);
    }
    Poly mulT(Poly b) const {
        if (b.size() == 0) {
            return Poly();
        }
        int n = b.size();
        std::reverse(b.a.begin(), b.a.end());
        return ((*this) * b).divxk(n - 1);
    }
    std::vector<Z> eval(std::vector<Z> x) const {
        if (size() == 0) {
            return std::vector<Z>(x.size(), 0);
        }
        const int n = std::max(int(x.size()), size());
        std::vector<Poly> q(4 * n);
        std::vector<Z> ans(x.size());
        x.resize(n);
        std::function<void(int, int, int)> build = [&](int p, int l, int r) {
            if (r - l == 1) {
                q[p] = Poly{1, -x[l]};
            } else {
                int m = (l + r) / 2;
                build(2 * p, l, m);
                build(2 * p + 1, m, r);
                q[p] = q[2 * p] * q[2 * p + 1];
            }
        };
        build(1, 0, n);
        std::function<void(int, int, int, const Poly &)> work = [&](int p, int l, int r, const Poly &num) {
            if (r - l == 1) {
                if (l < int(ans.size())) {
                    ans[l] = num[0];
                }
            } else {
                int m = (l + r) / 2;
                work(2 * p, l, m, num.mulT(q[2 * p + 1]).modxk(m - l));
                work(2 * p + 1, m, r, num.mulT(q[2 * p]).modxk(r - m));
            }
        };
        work(1, 0, n, mulT(q[1].inv(n)));
        return ans;
    }
};
```
